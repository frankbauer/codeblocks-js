import de.fau.tf.lgdv.CodeBlocks;
import de.fau.tf.lgdv.json.JsonSerializer;

class Foo implements JsonSerializer{
        public String toJson(){
            return "{\"test\":42}";
        }
}
public class App {
    public static void main(String[] args){
        System.out.println("Hello World {:test}");
        System.out.println("or anyone else {:checked}");
        int[] testArray = {1,2,3,4,5};
        System.out.println(testArray[2]);
        System.out.println(testArray[-1]);
        CodeBlocks.postResult(new Foo());
    }
}

class MainOverride {
    public static void main(String[] args){
        System.out.println("MainOverride was called");
        App.main(args);        
    }
}
