App 1 (with overlay)

function sum(values) {
    let total = 0
    for (let i = 0; i <= values.length; i++) {
        total += values[i]
    }
    return total
}

const data = [3, 4, 5]
console.log(sum(data))
console.log(Math.max(...data))
